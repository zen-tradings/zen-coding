export const TypingTerminal = () => {
  const commands = [
    "pi install git:github.com/zen-tradings/zen-coding",
    "npm run agent",
    "/zen plan",
  ]
  const [commandIndex, setCommandIndex] = useState(0)
  const [characterIndex, setCharacterIndex] = useState(0)
  const [deleting, setDeleting] = useState(false)
  const command = commands[commandIndex]

  useEffect(() => {
    if (window.matchMedia("(prefers-reduced-motion: reduce)").matches) {
      setCharacterIndex(command.length)
      return
    }

    const atEnd = characterIndex === command.length
    const atStart = characterIndex === 0
    const delay = deleting ? 35 : atEnd ? 1400 : 70
    const timer = window.setTimeout(() => {
      if (atEnd && !deleting) {
        setDeleting(true)
      } else if (atStart && deleting) {
        setDeleting(false)
        setCommandIndex((current) => (current + 1) % commands.length)
      } else {
        setCharacterIndex((current) => current + (deleting ? -1 : 1))
      }
    }, delay)

    return () => window.clearTimeout(timer)
  }, [characterIndex, command, deleting])

  return (
    <div className="overflow-hidden rounded-2xl border border-slate-700/80 bg-slate-950 shadow-2xl shadow-slate-950/20">
      <div className="flex items-center gap-2 border-b border-slate-800 px-4 py-3" aria-hidden="true">
        <span className="h-2.5 w-2.5 rounded-full bg-red-400"></span>
        <span className="h-2.5 w-2.5 rounded-full bg-amber-300"></span>
        <span className="h-2.5 w-2.5 rounded-full bg-emerald-400"></span>
        <span className="ml-2 font-mono text-xs text-slate-500">zen-coding</span>
      </div>
      <div className="flex min-h-20 items-center px-5 py-5 font-mono text-sm sm:text-base" aria-live="polite">
        <span className="mr-3 select-none text-cyan-400">❯</span>
        <span className="text-slate-100">{command.slice(0, characterIndex)}</span>
        <span className="ml-1 inline-block h-5 w-px animate-pulse bg-slate-200" aria-hidden="true"></span>
      </div>
    </div>
  )
}
